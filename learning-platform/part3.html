CREATE EXTENSION IF NOT EXISTS "pgcrypto";  -- for gen_random_uuid()
CREATE EXTENSION IF NOT EXISTS "vector";    -- for vector type + operators

-- USERS TABLE
CREATE TABLE users (
    id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    -- UUID: 128-bit random ID. Better than serial/integer because:
    -- (1) no sequential guessing by attackers
    -- (2) safe for distributed systems
    -- (3) can be generated client-side without DB round-trip
    email      VARCHAR(255) UNIQUE NOT NULL,
    name       VARCHAR(255),
    plan       VARCHAR(50) DEFAULT 'free',
    created_at TIMESTAMP DEFAULT NOW()
);

-- USER_INPUTS
CREATE TABLE user_inputs (
    id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id    UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    input_text TEXT NOT NULL,
    input_type VARCHAR(50) NOT NULL,
    metadata   JSONB DEFAULT '{}',  -- JSONB: binary JSON, indexable, queryable
    ip_address INET,                -- PostgreSQL native IP type
    created_at TIMESTAMP DEFAULT NOW()
);

-- PREDICTIONS
CREATE TABLE predictions (
    id               UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    input_id         UUID NOT NULL REFERENCES user_inputs(id) ON DELETE CASCADE,
    model_name       VARCHAR(100) NOT NULL,
    model_version    VARCHAR(20) NOT NULL,
    prediction_label VARCHAR(100),  -- 'HOT' | 'WARM' | 'COLD'
    confidence       FLOAT,
    raw_output       JSONB DEFAULT '{}',
    latency_ms       INTEGER,
    created_at       TIMESTAMP DEFAULT NOW()
);

-- EMBEDDINGS: vector representations for semantic search
CREATE TABLE embeddings (
    id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    input_id   UUID NOT NULL REFERENCES user_inputs(id) ON DELETE CASCADE,
    user_id    UUID NOT NULL REFERENCES users(id),
    embedding  vector(384),  -- 384 floats = all-MiniLM-L6-v2 output dimension
    model_name VARCHAR(100) DEFAULT 'all-MiniLM-L6-v2',
    created_at TIMESTAMP DEFAULT NOW()
);

-- B-tree indexes for filtering
CREATE INDEX idx_user_inputs_user_id ON user_inputs(user_id);
CREATE INDEX idx_user_inputs_created ON user_inputs(created_at DESC);
CREATE INDEX idx_predictions_label   ON predictions(prediction_label);

-- IVFFlat vector index for approximate nearest neighbor search
-- lists = 100: number of clusters (rule of thumb: sqrt(n_rows))
CREATE INDEX idx_embeddings_vector ON embeddings
    USING ivfflat (embedding vector_cosine_ops) WITH (lists = 100);

Querying pgvector — finding K nearest neighbors
SELECT
    ui.input_text,
    p.prediction_label,
    1 - (e.embedding <=> $1::vector) AS similarity
FROM embeddings e
JOIN user_inputs ui ON ui.id = e.input_id
JOIN predictions p  ON p.input_id = ui.id
WHERE e.input_type = 'lead_message'
  AND e.user_id = $2::uuid
ORDER BY e.embedding <=> $1::vector  -- ORDER BY cosine distance (smaller = more similar)
LIMIT 5;

🅆 sentence-transformers + FAISS

Used in your AI Market Research Copilot. sentence-transformers converts text to semantic embeddings; FAISS performs lightning-fast nearest-neighbor search.

sentence-transformers — What It Does

all-MiniLM-L6-v2 maps any sentence to a 384-D vector where semantically similar sentences have vectors that are close together:

  • "I want to sell my coaching program online" → [0.12, -0.43, 0.87, ...] (384 floats)
  • "monetize my mentoring service on the internet" → [0.11, -0.45, 0.86, ...] (very close!)
  • TF-IDF gives these a score of 0 (no shared keywords). sentence-transformers gives cosine similarity ≈ 0.92.

Why all-MiniLM-L6-v2? 384 dims (small = fast), runs on CPU, ~22MB, free, open source, excellent quality for English text.

FAISS vector store pattern from your market research copilot
import faiss
import numpy as np
from sentence_transformers import SentenceTransformer

model = SentenceTransformer("all-MiniLM-L6-v2")

class FAISSVectorStore:
    def __init__(self, dim: int = 384):
        # IndexFlatIP: exact inner product search
        # = cosine similarity when embeddings are normalized
        self.index = faiss.IndexFlatIP(dim)
        self.texts = []      # parallel list: texts[i] = vector i
        self.metadata = []

    def add_texts(self, texts: list[str], metadata: list[dict]):
        embeddings = model.encode(texts, normalize_embeddings=True)
        # FAISS requires float32 C-contiguous array
        embeddings = np.array(embeddings, dtype=np.float32)
        self.index.add(embeddings)
        self.texts.extend(texts)
        self.metadata.extend(metadata)

    def search(self, query: str, k: int = 5) -> list[dict]:
        query_emb = model.encode([query], normalize_embeddings=True)
        query_emb = np.array(query_emb, dtype=np.float32)

        # D = inner product scores (0-1 for normalized vectors)
        # I = indices into self.texts
        D, I = self.index.search(query_emb, k)

        results = []
        for score, idx in zip(D[0], I[0]):
            if idx == -1: continue  # no result
            results.append({
                "text": self.texts[idx],
                "score": float(score),
                "metadata": self.metadata[idx]
            })
        return results  # sorted by score descending

🧱 RAG Pipeline — Retrieval-Augmented Generation

RAG is the architectural pattern powering your AI Market Research Copilot. Instead of asking an LLM to answer from memory (hallucinations), you first retrieve relevant context from your own documents.

Why RAG?

Without RAG: "What are the top 3 competitors in our report?" → LLM makes up plausible-sounding but wrong answers.

With RAG: Find 5 most relevant chunks from your uploaded report → inject into LLM prompt → LLM generates answer grounded in actual document content.

Step 1+2: Document parsing and chunking
import PyPDF2

def parse_document(file_path: str, content_type: str) -> str:
    if content_type == "application/pdf":
        reader = PyPDF2.PdfReader(file_path)
        return "\n".join(page.extract_text() for page in reader.pages)
    else:
        return open(file_path).read()

# Chunking with overlap: important context at chunk boundaries is not lost
def chunk_text(text: str, chunk_size: int = 500, overlap: int = 50) -> list[str]:
    chunks = []
    start = 0
    while start < len(text):
        end = start + chunk_size
        chunk = text[start:end]
        if chunk.strip():
            chunks.append(chunk)
        start = end - overlap  # move forward by (chunk_size - overlap)
    return chunks
Step 5+6: Retrieval + augmented generation (RAG core)
import google.generativeai as genai

genai.configure(api_key=os.getenv("GEMINI_API_KEY"))
llm = genai.GenerativeModel("gemini-1.5-flash")

def rag_answer(query: str, store: FAISSVectorStore, k: int = 5) -> dict:
    results = store.search(query, k=k)

    context = "\n\n---\n\n".join(
        f"[Source chunk {i+1}]:\n{r['text']}"
        for i, r in enumerate(results)
    )

    prompt = f"""You are a market research analyst. Answer the question using ONLY
the provided context. If the answer is not in the context, say so.

CONTEXT:
{context}

QUESTION: {query}

Provide a structured answer with specific evidence from the context."""

    response = llm.generate_content(prompt)
    return {
        "answer": response.text,
        "sources": [r["text"][:200] for r in results],
        "num_chunks_retrieved": len(results)
    }

Strategy + Factory + ABC Pattern

The Strategy pattern defines a family of algorithms, encapsulates each, and makes them interchangeable. BaseLLMClient is the interface; ClaudeClient and OpenAIClient are concrete strategies.

Factory + Strategy + ABC from your code
from abc import ABC, abstractmethod

class BaseLLMClient(ABC):
    @abstractmethod
    async def generate_review(self, prompt: str) -> str: ...

    @abstractmethod
    async def parse_review_response(self, raw: str) -> dict: ...

class ClaudeClient(BaseLLMClient):
    async def generate_review(self, prompt: str) -> str:
        # Anthropic-specific with prompt caching
        ...

class OpenAIClient(BaseLLMClient):
    async def generate_review(self, prompt: str) -> str:
        # OpenAI-specific with JSON mode
        ...

def _build_llm_client() -> BaseLLMClient:
    provider = get_settings().llm_provider
    if provider == "openai":
        return OpenAIClient()
    return ClaudeClient()  # default

class ReviewService:
    def __init__(self):
        self._llm = _build_llm_client()  # never knows which provider

    async def run_review(self, ...):
        raw = await self._llm.generate_review(prompt)

3-Layer Fallback Strategy

  • Layer 1 — Retry with exponential backoff: Try same provider 3x with delays 1s, 2s, 4s.
  • Layer 2 — Provider fallback: If primary exhausts retries, switch to backup provider.
  • Layer 3 — Job queue: If all providers fail, push to SQS/BullMQ. User gets: "Content queued, we'll notify you."
3-layer fallback (JavaScript pseudocode)
async function generateWithFallback(request) {
  const providers = [primaryProvider, backupProvider];

  for (const provider of providers) {
    for (let attempt = 1; attempt <= 3; attempt++) {
      try {
        return await withTimeout(provider.generate(request), 30_000);
      } catch (err) {
        if (attempt < 3) {
          await sleep(attempt * 1000);  // 1s, 2s
        }
      }
    }
  }

  // Layer 3: all providers exhausted
  await jobQueue.add('generate', request, { delay: 60_000 });
  return { status: 'queued', message: 'Content queued. You will be notified.' };
}

How Streamlit Works

Every time a user interacts, Streamlit re-runs the entire script from top to bottom. State that must persist across reruns lives in st.session_state.

⚡ Python Async/Await

async/await — complete mental model
import asyncio

async def fetch_data(url: str) -> dict:
    # await suspends THIS coroutine, giving up the event loop
    # to run other coroutines while waiting for the HTTP response.
    # The thread is NOT blocked.
    response = await httpx.AsyncClient().get(url)
    return response.json()

# Running multiple tasks CONCURRENTLY
async def main():
    # asyncio.gather runs all 3 coroutines concurrently
    # Total time = max(t1, t2, t3), NOT t1+t2+t3
    results = await asyncio.gather(
        fetch_data("https://api1.example.com"),
        fetch_data("https://api2.example.com"),
        fetch_data("https://api3.example.com"),
    )

# Common mistake:
async def bad():
    time.sleep(5)  # WRONG: blocks the entire event loop!

async def good():
    await asyncio.sleep(5)  # yields control while sleeping
