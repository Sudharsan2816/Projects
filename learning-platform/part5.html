from sentence_transformers import SentenceTransformer
import psycopg2
import anthropic

embed_model = SentenceTransformer("all-MiniLM-L6-v2")
llm = anthropic.Anthropic()

def rag_query(question: str, top_k: int = 5) -> str:
    # 1. Embed the question
    q_embedding = embed_model.encode(question).tolist()

    # 2. Vector search (pgvector)
    conn = psycopg2.connect(DATABASE_URL)
    with conn.cursor() as cur:
        cur.execute("""
            SELECT content, source, 1 - (embedding <=> %s::vector) AS similarity
            FROM documents
            ORDER BY embedding <=> %s::vector
            LIMIT %s
        """, (q_embedding, q_embedding, top_k))
        chunks = cur.fetchall()

    # 3. Build context from retrieved chunks
    context = "\n\n".join([
        f"[Source: {chunk[1]}]\n{chunk[0]}"
        for chunk in chunks
    ])

    # 4. Augmented prompt
    prompt = f"""Answer the question based on these retrieved documents:

{context}

Question: {question}
Answer (if not in documents, say "I don't have that information"):"""

    # 5. Generate answer
    response = llm.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}]
    )
    return response.content[0].text

Prompt Engineering Best Practices

  • Be explicit about output format: "Return JSON with keys: name, score, reason"
  • Use few-shot examples: Show 2-3 examples of the exact transformation you want
  • Chain of thought: "Think step by step before giving the final answer"
  • Persona: "You are a senior Python engineer at a FAANG company"
  • Constraints: "Be concise. Max 3 sentences. No markdown."
  • Negative prompting: "Do NOT make up information. If unsure, say 'I don't know'"
  • System vs user split: Put stable instructions in system, variable input in user turn
  • Temperature: 0-0.3 for structured tasks, 0.7+ for creative writing

🧠 Quiz: AI Engineering

1. In RAG evaluation, what does "Faithfulness" measure?

✅ Correct! Faithfulness measures whether every claim in the answer can be traced back to the retrieved context. Low faithfulness = hallucination.
❌ Wrong. Faithfulness = answer claims that are supported by context / total answer claims. It's the anti-hallucination metric.

2. When should you use the "Plan-and-Execute" agent pattern vs "ReAct"?

✅ Correct! Plan-and-Execute: upfront planning → sequential execution (less adaptive but more predictable). ReAct: interleaved reasoning and acting (more flexible but can drift).
❌ Wrong. Use Plan-and-Execute for structured multi-step tasks (e.g., report generation). Use ReAct for dynamic tool-use scenarios where the next step depends on tool results.

ML Monitoring — Data & Concept Drift

CRITICALModels degrade silently in production
from scipy import stats
import numpy as np

def detect_data_drift(reference_data, current_data, threshold=0.05):
    """
    Kolmogorov-Smirnov test for numerical feature drift.
    Returns drifted features with p-values below threshold.
    """
    drifted = {}
    for feature in reference_data.columns:
        stat, p_value = stats.ks_2samp(
            reference_data[feature].dropna(),
            current_data[feature].dropna()
        )
        if p_value < threshold:
            drifted[feature] = {"ks_stat": stat, "p_value": p_value}
    return drifted

# Types of drift:
# 1. Data drift (covariate shift): P(X) changes, P(Y|X) same → feature distribution changed
# 2. Concept drift: P(Y|X) changes → relationship between features and target changed
# 3. Label drift: P(Y) changes → class distribution changed

# Monitoring checklist:
# ✅ Input feature distributions (daily histogram comparison)
# ✅ Prediction score distributions
# ✅ Model accuracy/F1 (requires ground truth labels)
# ✅ Latency p50, p95, p99
# ✅ Error rates (null predictions, out-of-range outputs)
# ✅ Business metrics (click-through rate, conversion)

🧠 Quiz: MLOps

3. What is the difference between "data drift" and "concept drift"?

✅ Correct! Data drift = your users changed (features look different). Concept drift = the world changed (same features now mean different outcomes). Both cause model degradation.
❌ Wrong. Data drift = P(X) changes (feature distributions shift). Concept drift = P(Y|X) changes (the same feature values now predict differently). Both need monitoring.

import torch
import torch.nn as nn
import torch.optim as optim

# Basic feedforward network
class MLP(nn.Module):
    def __init__(self, input_dim, hidden_dims, output_dim, dropout=0.3):
        super().__init__()
        layers = []
        dims = [input_dim] + hidden_dims
        for i in range(len(dims)-1):
            layers.extend([
                nn.Linear(dims[i], dims[i+1]),
                nn.BatchNorm1d(dims[i+1]),
                nn.ReLU(),
                nn.Dropout(dropout)
            ])
        layers.append(nn.Linear(dims[-1], output_dim))
        self.network = nn.Sequential(*layers)

    def forward(self, x):
        return self.network(x)

# Training loop
model = MLP(input_dim=128, hidden_dims=[256, 128, 64], output_dim=10)
optimizer = optim.AdamW(model.parameters(), lr=1e-3, weight_decay=1e-4)

Transformer Architecture

KEY CONCEPTThe architecture behind GPT, BERT, Claude, etc.

Transformers replaced RNNs via self-attention: every token attends to every other token in parallel, enabling training on GPUs at scale.

import torch
import torch.nn as nn
import math

class MultiHeadSelfAttention(nn.Module):
    def __init__(self, d_model, num_heads):
        super().__init__()
        self.num_heads = num_heads
        self.d_k = d_model // num_heads
        self.W_q = nn.Linear(d_model, d_model)
        self.W_k = nn.Linear(d_model, d_model)
        self.W_v = nn.Linear(d_model, d_model)
        self.W_o = nn.Linear(d_model, d_model)

    def forward(self, x, mask=None):
        B, T, D = x.shape
        Q = self.W_q(x).view(B, T, self.num_heads, self.d_k).transpose(1, 2)
        K = self.W_k(x).view(B, T, self.num_heads, self.d_k).transpose(1, 2)
        V = self.W_v(x).view(B, T, self.num_heads, self.d_k).transpose(1, 2)

        # Scaled dot-product attention
        scores = torch.matmul(Q, K.transpose(-2, -1)) / math.sqrt(self.d_k)
        if mask is not None:
            scores = scores.masked_fill(mask == 0, -1e9)
        attn = torch.softmax(scores, dim=-1)

        out = torch.matmul(attn, V)
        out = out.transpose(1, 2).contiguous().view(B, T, D)
        return self.W_o(out)
ℹ️ Why attention works "Attention is all you need": instead of processing sequences step-by-step (RNN), attention computes all token relationships simultaneously. O(n²) memory but fully parallelizable → orders of magnitude faster training.

🧠 Quiz: Deep Learning

4. Why did Transformers replace RNNs for NLP tasks?

✅ Correct! RNNs process tokens sequentially (bottleneck for parallelization) and struggle with long-range dependencies due to vanishing gradients. Transformers: parallel computation + attention that reaches any distance.
❌ Wrong. The key advantage: Transformers parallelize over the entire sequence (unlike RNNs' sequential processing) and attention directly connects any two tokens regardless of distance.

URL Shortener (Classic Problem)

DESIGN PROBLEMDesign bit.ly / tinyurl.com

Requirements: 100M URLs/day writes, 10B reads/day, 7-year retention, p99 redirect <10ms

import hashlib
import base64
import redis
import psycopg2

# Short code generation — Base62 encoding of MD5 hash
def generate_short_code(long_url: str, length: int = 7) -> str:
    # Take first 43 bits of MD5 → base62 encode
    md5 = hashlib.md5(long_url.encode()).digest()
    # Convert bytes to base62
    alphabet = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
    number = int.from_bytes(md5[:6], 'big')
    code = ""
    while number:
        code = alphabet[number % 62] + code
        number //= 62
    return code[:length]  # "xK7mP2z"

# High-performance redirect flow:
# 1. Check Redis cache (hot URLs) → O(1)
# 2. If miss → query DB → cache result with TTL
# 3. Return 301 (cached by browser) or 302 (analytics tracking)

class URLShortener:
    def __init__(self):
        self.redis = redis.Redis(host="redis", decode_responses=True)
        self.db = psycopg2.connect(DATABASE_URL)

    def redirect(self, code: str) -> str:
        # 1. Cache hit
        long_url = self.redis.get(f"url:{code}")
        if long_url:
            return long_url

        # 2. DB lookup
        with self.db.cursor() as cur:
            cur.execute("SELECT long_url FROM urls WHERE code = %s", (code,))
            row = cur.fetchone()
            if not row:
                raise KeyError("URL not found")
            long_url = row[0]

        # 3. Cache with 24h TTL
        self.redis.setex(f"url:{code}", 86400, long_url)
        return long_url
💡 Scale decisions 301 vs 302: Use 302 to keep analytics (server sees each redirect). Use 301 for CDN caching (browser caches, no server hit). Sharding: Range-based on code prefix vs hash-based. Hash = even distribution. Store full URL in object storage for very long URLs.

Design a Rate Limiter

import redis
import time

class TokenBucketRateLimiter:
    """
    Token Bucket: allows burst traffic up to bucket capacity.
    Tokens refill at a fixed rate. Best for APIs with bursty traffic.
    """
    def __init__(self, redis_client, capacity: int, refill_rate: float):
        self.redis = redis_client
        self.capacity = capacity       # Max burst: 100 requests
        self.refill_rate = refill_rate # Tokens/second: 10 req/s sustained

    def is_allowed(self, user_id: str) -> bool:
        key = f"rate:{user_id}"
        now = time.time()
        pipe = self.redis.pipeline()

        pipe.hget(key, "tokens")
        pipe.hget(key, "last_refill")
        tokens_str, last_refill_str = pipe.execute()

        tokens = float(tokens_str or self.capacity)
        last_refill = float(last_refill_str or now)

        # Refill tokens based on elapsed time
        elapsed = now - last_refill
        tokens = min(self.capacity, tokens + elapsed * self.refill_rate)

        if tokens < 1:
            return False  # Rate limited!

        # Consume one token
        self.redis.hset(key, mapping={"tokens": tokens - 1, "last_refill": now})
        self.redis.expire(key, 3600)
        return True

# Sliding Window Counter (more accurate, slightly more memory)
class SlidingWindowRateLimiter:
    def __init__(self, redis_client, limit: int, window_seconds: int):
        self.redis = redis_client
        self.limit = limit
        self.window = window_seconds

    def is_allowed(self, user_id: str) -> bool:
        key = f"sw:{user_id}"
        now = time.time()
        window_start = now - self.window

        pipe = self.redis.pipeline()
        pipe.zremrangebyscore(key, 0, window_start)  # Remove old requests
        pipe.zadd(key, {str(now): now})               # Add current request
        pipe.zcard(key)                                # Count in window
        pipe.expire(key, self.window)
        _, _, count, _ = pipe.execute()

        return count <= self.limit

CAP Theorem

FUNDAMENTALIn a distributed system: choose 2 of 3
PropertyMeaningExample
Consistency (C)Every read gets the most recent writeBank balance: always correct
Availability (A)Every request gets a response (not an error)Always responds, might be stale
Partition Tolerance (P)Works despite network partitionsRequired in any distributed system
⚠️ The Real Choice Since network partitions WILL happen, you're always choosing between C and A during a partition: CA systems (SQL): Consistent + available, halt on partitions → Postgres, MySQL CP systems: Consistent when partitioned, sacrifice availability → HBase, Zookeeper, etcd AP systems: Available when partitioned, sacrifice consistency → Cassandra, DynamoDB, CouchDB

🧠 Quiz: System Design

5. Twitter has a celebrity with 50M followers. They post a tweet. Why is "fan-out on write" problematic here?

✅ Correct! 1 tweet → 50M write operations is write amplification. Even at 100K writes/second, it takes 500 seconds (8+ minutes). Twitter uses a hybrid: push for regular users, pull for celebrities.
❌ Wrong. Fan-out on write for 50M followers = 50M Redis/DB writes per tweet. This causes massive spike and delays. The solution: pull-on-read for high-follower accounts.

6. Cassandra is described as an AP system. What does this mean in practice?

✅ Correct! AP = Available + Partition-tolerant. During a split-brain scenario, Cassandra keeps responding (no downtime) but nodes may have different versions of the data. It reconciles later (eventual consistency).
❌ Wrong. AP systems prioritize availability over consistency. Cassandra uses eventual consistency — all nodes converge to the same value eventually, but reads during partitions may be stale.

7. Your app needs to serve ML predictions to 10M users globally with <50ms latency. Which AWS services are most critical?

✅ Correct! For global low-latency ML: CloudFront to cache predictions, ElastiCache for hot results, SageMaker with multi-region endpoints, Auto Scaling for demand spikes.

8. Under the "Reliability" pillar, what is the purpose of the Circuit Breaker pattern?

✅ Correct! Circuit breaker: CLOSED (normal) → OPEN (failing service, fail fast) → HALF-OPEN (test if recovered). Prevents one slow service from blocking your entire system.
❌ Wrong. Circuit breaker prevents cascading failures. When a service fails repeatedly, the circuit "opens" and requests fail fast without waiting — preventing timeout pile-ups from bringing down everything.

import langfuse
from functools import wraps

# LangFuse for LLM observability
lf = langfuse.Langfuse(
    public_key="pk-...",
    secret_key="sk-...",
    host="https://cloud.langfuse.com"
)

def trace_llm_call(func):
    @wraps(func)
    async def wrapper(*args, **kwargs):
        trace = lf.trace(name=func.__name__)
        generation = trace.generation(
            name="claude-call",
            model="claude-sonnet-4-6",
            input=kwargs.get("prompt")
        )
        try:
            result = await func(*args, **kwargs)
            generation.end(output=result, usage={"input": 100, "output": 50})
            return result
        except Exception as e:
            generation.end(level="ERROR", status_message=str(e))
            raise
    return wrapper

🧠 Quiz: DevOps for AI

9. In a Kubernetes HPA (Horizontal Pod Autoscaler), what does averageUtilization: 70 on CPU mean?

✅ Correct! HPA scales replicas to maintain average CPU utilization near 70% of the requested CPU. If average goes to 140%, HPA doubles the pod count. If it drops to 35%, HPA scales down.
❌ Wrong. HPA continuously adjusts replica count to keep average CPU across all pods near the target utilization. Scale up when exceeding target, scale down when below.

10. What is the main advantage of using LangFuse for LLM observability over generic tools like Datadog?

✅ Correct! LangFuse is purpose-built for LLM apps: traces full chains, logs prompts/responses, tracks token costs per user, and allows attaching quality scores. Generic APM tools miss this LLM-specific context.
❌ Wrong. LangFuse is LLM-native: it understands chains, tool calls, generation metadata, and evaluation. Generic tools only see HTTP requests, not the semantics of LLM interactions.
